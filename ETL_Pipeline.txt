CREATE OR REFRESH STREAMING TABLE bronze_hotel_bookings AS
SELECT 
    *,
    _metadata.file_name AS source_file,
    current_timestamp() AS loaded_at
FROM STREAM read_files(
    '/Volumes/hospitality_catalog/bookings/raw_files/',
    format => 'csv',
    header => true
);

-- 1. Silver Clean Table
CREATE OR REFRESH MATERIALIZED VIEW silver_bookings_clean (
    CONSTRAINT valid_stay_dates EXPECT (check_out_date > check_in_date) ON VIOLATION DROP ROW,
    CONSTRAINT valid_room_rate EXPECT (room_rate >= 0) ON VIOLATION DROP ROW
) AS
SELECT DISTINCT
    booking_id,
    guest_id,
    CAST(booking_date AS DATE) AS booking_date,
    CAST(check_in_date AS DATE) AS check_in_date,
    CAST(check_out_date AS DATE) AS check_out_date,
    COALESCE(CAST(nights AS INT), DATEDIFF(check_out_date, check_in_date)) AS nights,
    INITCAP(TRIM(hotel_name)) AS hotel_name,
    COALESCE(INITCAP(TRIM(city)), 'Unknown') AS city,
    INITCAP(TRIM(region)) AS region,
    INITCAP(TRIM(room_type)) AS room_type,
    COALESCE(INITCAP(TRIM(booking_channel)), 'Unknown') AS booking_channel,
    CAST(guests AS INT) AS guests,
    CAST(room_rate AS DOUBLE) AS room_rate,
    COALESCE(
        CAST(total_amount AS DOUBLE),
        COALESCE(CAST(nights AS INT), DATEDIFF(check_out_date, check_in_date)) * CAST(room_rate AS DOUBLE)
    ) AS total_amount,
    COALESCE(CAST(discount_amount AS DOUBLE), 0.0) AS discount_amount,
    INITCAP(TRIM(meal_plan)) AS meal_plan,
    payment_method,
    payment_status,
    booking_status,
    CAST(guest_rating AS DOUBLE) AS guest_rating
FROM bronze_hotel_bookings;

-- 2. Silver Rejected Table
CREATE OR REFRESH MATERIALIZED VIEW silver_bookings_rejected AS
SELECT 
    *,
    CASE 
        WHEN check_out_date <= check_in_date THEN 'Check-out is not after check-in'
        WHEN room_rate < 0 THEN 'Room rate is negative'
        ELSE 'Other invalid rule'
    END AS reject_reason
FROM bronze_hotel_bookings
WHERE check_out_date <= check_in_date OR room_rate < 0;


-- 1. Monthly Performance (BR-01)
CREATE OR REFRESH MATERIALIZED VIEW gold_monthly_bookings AS
SELECT 
    DATE_TRUNC('MONTH', check_in_date) AS stay_month,
    COUNT(*) AS bookings,
    SUM(CASE WHEN booking_status = 'Cancelled' THEN 1 ELSE 0 END) AS cancelled_bookings,
    SUM(CASE WHEN booking_status = 'Completed' THEN nights ELSE 0 END) AS room_nights,
    SUM(CASE WHEN booking_status = 'Completed' THEN (total_amount - discount_amount) ELSE 0 END) AS net_revenue
FROM silver_bookings_clean
GROUP BY DATE_TRUNC('MONTH', check_in_date);

-- 2. Hotel Performance (BR-02)
CREATE OR REFRESH MATERIALIZED VIEW gold_hotel_summary AS
SELECT 
    region,
    city,
    hotel_name,
    COUNT(*) AS bookings,
    SUM(CASE WHEN booking_status = 'Completed' THEN nights ELSE 0 END) AS room_nights,
    SUM(CASE WHEN booking_status = 'Completed' THEN (total_amount - discount_amount) ELSE 0 END) AS net_revenue,
    ROUND(
        SUM(CASE WHEN booking_status = 'Completed' THEN (total_amount - discount_amount) ELSE 0 END) / 
        NULLIF(SUM(CASE WHEN booking_status = 'Completed' THEN nights ELSE 0 END), 0), 
        2
    ) AS avg_daily_rate
FROM silver_bookings_clean
GROUP BY region, city, hotel_name;

-- 3. Room Type Performance (BR-03)
CREATE OR REFRESH MATERIALIZED VIEW gold_room_type_summary AS
SELECT 
    room_type,
    COUNT(*) AS bookings,
    SUM(CASE WHEN booking_status = 'Completed' THEN nights ELSE 0 END) AS room_nights,
    SUM(CASE WHEN booking_status = 'Completed' THEN (total_amount - discount_amount) ELSE 0 END) AS net_revenue,
    ROUND(AVG(guest_rating), 2) AS avg_rating
FROM silver_bookings_clean
GROUP BY room_type;

-- 4. Channel Summary (BR-04)
CREATE OR REFRESH MATERIALIZED VIEW gold_channel_summary AS
SELECT 
    booking_channel,
    COUNT(*) AS bookings,
    SUM(CASE WHEN booking_status = 'Cancelled' THEN 1 ELSE 0 END) AS cancelled_bookings,
    ROUND(SUM(CASE WHEN booking_status = 'Cancelled' THEN 1.0 ELSE 0.0 END) * 100.0 / COUNT(*), 2) AS cancellation_rate_pct,
    SUM(CASE WHEN booking_status = 'Completed' THEN (total_amount - discount_amount) ELSE 0 END) AS net_revenue
FROM silver_bookings_clean
GROUP BY booking_channel;

-- 5. Data Quality & Reconciliation (BR-05 & BR-06)
CREATE OR REFRESH MATERIALIZED VIEW gold_data_quality AS
SELECT 
    (SELECT COUNT(*) FROM bronze_hotel_bookings) AS bronze_rows,
    (SELECT COUNT(*) FROM silver_bookings_clean) AS clean_rows,
    (SELECT COUNT(*) FROM silver_bookings_rejected) AS rejected_rows,
    ((SELECT COUNT(*) FROM bronze_hotel_bookings) - (SELECT COUNT(DISTINCT booking_id) FROM bronze_hotel_bookings)) AS duplicate_rows,
    CASE 
        WHEN (SELECT COUNT(*) FROM bronze_hotel_bookings) = 
             ((SELECT COUNT(*) FROM silver_bookings_clean) + 
              (SELECT COUNT(*) FROM silver_bookings_rejected) + 
              ((SELECT COUNT(*) FROM bronze_hotel_bookings) - (SELECT COUNT(DISTINCT booking_id) FROM bronze_hotel_bookings)))
        THEN 'PASSED' ELSE 'FAILED'
    END AS reconciliation_status;
